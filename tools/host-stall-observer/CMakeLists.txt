cmake_minimum_required(VERSION 3.16)
project(HostStallObserver C)
if(WIN32)
  add_executable(host_stall_observer observe.c)
  target_link_libraries(host_stall_observer PRIVATE dbghelp psapi)
  add_executable(host_stall_fixture fixture.c)
  if(MSVC)
    target_link_options(host_stall_fixture PRIVATE /MAP /DEBUG)
    target_compile_options(host_stall_fixture PRIVATE /Zi)
  endif()
else()
  message(STATUS "Windows process observer omitted on this platform; no runtime dependency")
endif()
